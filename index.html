/** * AI 영업 학습 & 작성 도우미 - Server Script (Code.gs) */ // 1. 웹 앱 접속 시 Index.html 화면 출력 function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('AI 영업 학습 & 작성 도우미') .setXframeOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } /** * 2. Gemini API(gemini-3.5-flash)를 호출하여 영업 스크립트 생성 * @param {string} prompt - 사용자 요청 사항 * @param {string} category - 영업 시나리오 카테고리 * @returns {string} AI가 생성한 답변 텍스트 */ function generateSalesResponse(prompt, category) { try { const apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY'); if (!apiKey) { return "오류: GEMINI_API_KEY가 '스크립트 속성'에 설정되어 있지 않습니다. 프로젝트 설정을 확인해 주세요."; } const modelName = 'gemini-3.5-flash'; const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`; // 영업 카테고리별 프롬프트 가이드라인 설정 let systemInstruction = "당신은 전문 B2B/B2C 영업 컨설턴트 및 영업 코치입니다. 한국어로 논리적이고 설득력 있는 답변을 작성하세요."; if (category === 'email') { systemInstruction += " 고객의 반응을 이끌어낼 수 있는 세련되고 가독성 높은 영업 제안 이메일을 작성해 주세요."; } else if (category === 'objection') { systemInstruction += " 고객의 거절 이유 및 우려 사항에 공감하고, 논리적으로 반론을 극복할 수 있는 실전 대화 시나리오를 제시해 주세요."; } else if (category === 'proposal') { systemInstruction += " 제품 및 서비스의 핵심 가치 제안(Value Proposition)과 차별점을 강조한 요약 스크립트를 작성해 주세요."; } const fullPrompt = `${systemInstruction}\n\n[요청 사항]\n${prompt}`; const payload = { contents: [{ parts: [{ text: fullPrompt }] }], generationConfig: { temperature: 0.7, maxOutputTokens: 2048 } }; const options = { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true }; const response = UrlFetchApp.fetch(url, options); const statusCode = response.getResponseCode(); const responseText = response.getContentText(); if (statusCode !== 200) { Logger.log(`API 오류 [HTTP ${statusCode}]: ${responseText}`); return `API 요청 실패 (HTTP ${statusCode}): ${responseText}`; } const json = JSON.parse(responseText); if (json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts) { return json.candidates[0].content.parts[0].text; } else { return "AI 응답을 생성할 수 없거나 응답 형식이 올바르지 않습니다."; } } catch (error) { Logger.log("실행 중 오류 발생: " + error.toString()); return "스크립트 실행 에러: " + error.message; } } /** * 3. 구글 시트에 스크립트 기록 자동 저장 */ function saveToSheet(category, prompt, scriptResult) { try { let sheetId = PropertiesService.getScriptProperties().getProperty('SPREADSHEET_ID'); let ss; if (sheetId) { ss = SpreadsheetApp.openById(sheetId); } else { ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { ss = SpreadsheetApp.create('AI 영업 스크립트 저장소'); PropertiesService.getScriptProperties().setProperty('SPREADSHEET_ID', ss.getId()); } } let sheet = ss.getActiveSheet(); // 시트에 헤더가 없을 경우 자동 추가 if (sheet.getLastRow() === 0) { sheet.appendRow(['작성 일시', '카테고리', '요청 사항', 'AI 생성 스크립트']); sheet.getRange(1, 1, 1, 4).setFontWeight('bold').setBackground('#e2e8f0'); } const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss'); sheet.appendRow([timestamp, category, prompt, scriptResult]); return { success: true, message: '구글 시트에 성공적으로 저장되었습니다!', url: ss.getUrl() }; } catch (error) { return { success: false, message: '시트 저장 실패: ' + error.message }; } } /** * 4. 생성된 스크립트를 이메일로 발송 */ function sendScriptEmail(targetEmail, subject, scriptResult) { try { const email = targetEmail || Session.getActiveUser().getEmail(); if (!email) { return { success: false, message: '수신 이메일 주소를 확인할 수 없습니다.' }; } const emailSubject = subject ? `[AI 영업 도우미] ${subject}` : '[AI 영업 도우미] 생성된 영업 스크립트'; const emailBody = `안녕하세요,\n\nAI 영업 작성 도우미에서 생성한 스크립트입니다.\n\n========================================\n\n${scriptResult}\n\n========================================\n\n감사합니다.`; MailApp.sendEmail(email, emailSubject, emailBody); return { success: true, message: `${email} 주소로 이메일이 발송되었습니다!` }; } catch (error) { return { success: false, message: '이메일 발송 실패: ' + error.message }; } }
